clc
clearvars
close all

f=input('dy/dx:');
x0=input('enter the initial value x0:');
y0=input('enter the initial value y0:');
xg=input('enter the function xg:');
h=input('enter the given function h:');
n=(xg-x0)/h;

for i=1:n
    k1=h*f(x0,y0);
    k2=h*f(x0+h/2,y0+k1/2);
    k3=h*f(x0+h/2,y0+k2/2);
    k4=h*f(x0+h,y0+k3);
    k=(k1+2*k2+2*k3+k4)/6;

    x0=x0+h;
    yg=y0+k;
    y0=yg;

    fprintf('x= %.4f, yg= %.4f\n',x0,yg);
end
