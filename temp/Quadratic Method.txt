%Quadratic Method
clc;
clear all;
close all;
X=input('X:');
Y=input('Y:');
N=input('N:');
sumX=0;
sumY=0;
sumXY=0;
sumX2=0;
sumX2Y=0;
sumX3=0;
sumX4=0;
for i=1:N
    sumX=sumX+X(i);
    sumY=sumY+Y(i);
    sumXY=sumXY+X(i)*Y(i);
    sumX2=sumX2+X(i)^2;
    sumX2Y=sumX2Y+X(i)^2*Y(i);
    sumX3=sumX3+X(i)^3;
    sumX4=sumX4+X(i)^4;
end
A=[sumX2 sumX N; sumX3 sumX2 sumX; 
    sumX4 sumX3 sumX2];
B=[sumY; sumXY; sumX2Y];
C=inv(A)*(B);
fprintf('C=%.4f\n', C)
