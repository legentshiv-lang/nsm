%Straight line Method
clc;
clear all;
close all;
X=input('X:');
Y=input('Y:');
N=input('N:');
sumX=0;
sumY=0;
sumX2=0;
sumXY=0;
for i=1:N
    sumX=sumX+X(i);
    sumY=sumY+Y(i);
    sumX2=sumX2+X(i)^2;
    sumXY=sumXY+X(i)*Y(i);
end
A=[sumX N; sumX2 sumX];
B=[sumY; sumXY];
C=inv(A)*(B);
fprintf('C=%.4f\n', C)
