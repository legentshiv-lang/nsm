%Inverse Lagrange's Method
clc;
clear all;
close all;
X=input('X:');
Y=input('Y:');
n=input('n:');
Yg=input('Yg:');
Xg=0;
for i=1:n
    L(i)=1;
    for j=1:n
        if i~=j
            L(i)=L(i)*(Yg-Y(j))/(Y(i)-Y(j));
        end
    end
    Xg=Xg+L(i)*X(i);
    fprintf('L(%d)=%.4f\n', i, L(i))
end
fprintf('Xg=%.4f',Xg)
